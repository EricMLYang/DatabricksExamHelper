SELECT 
    primary_table, 
    SUM(usage_quantity) as total_dbu,
    count(*) as op_count
FROM system.storage.predictive_optimization_operations_history
WHERE start_time > now() - interval 30 days
GROUP BY 1
ORDER BY 2 DESC;
